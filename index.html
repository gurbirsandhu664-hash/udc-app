document.getElementById('classifyBtn').addEventListener('click', async (e) => { e.preventDefault(); const titleInput = document.getElementById('titleInput'); const subjectInput = document.getElementById('subjectInput'); const resultDiv = document.getElementById('resultDiv'); const btn = document.getElementById('classifyBtn'); const title = titleInput ? titleInput.value.trim() : ''; const subject = subjectInput ? subjectInput.value.trim() : ''; if (!title) { alert("ਕਿਰਪਾ ਕਰਕੇ ਪਹਿਲਾਂ ਕਿਤਾਬ ਦਾ ਨਾਮ ਦਰਜ ਕਰੋ।"); return; } // ਕਲਿੱਕ ਹੋਣ 'ਤੇ ਬਟਨ ਤੁਰੰਤ ਬੰਦ ਕਰੋ ਤਾਂ ਜੋ ਡਬਲ-ਰਿਕੁਐਸਟ ਨਾ ਜਾਵੇ btn.disabled = true; btn.innerText = "ਵਰਗੀਕਰਨ ਕੀਤਾ ਜਾ ਰਿਹਾ ਹੈ..."; try { const response = await fetch('/api/classify', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ title, subject }) }); const resData = await response.json(); if (!response.ok || !resData.success) { throw new Error(resData.error || "ਵਰਗੀਕਰਨ ਕਰਨ ਵਿੱਚ ਅਸਫਲ ਰਿਹਾ"); } const { ddc, udc, summary } = resData.data; resultDiv.innerHTML = `
Gemini 2.5 Flash

DDC ਵਰਗੀਕਰਨ

ਨੰਬਰ: ${ddc?.number || 'N/A'}

ਵਿਸ਼ਾ: ${ddc?.heading || 'N/A'}

UDC ਵਰਗੀਕਰਨ

ਨੰਬਰ: ${udc?.number || 'N/A'}

ਵਿਸ਼ਾ: ${udc?.heading || 'N/A'}

ਵੇਰਵਾ: ${summary || 'ਕੋਈ ਵੇਰਵਾ ਨਹੀਂ ਮਿਲਿਆ'}

`; } catch (err) { resultDiv.innerHTML = `
ਗਲਤੀ: ${err.message}
`; } finally { // ਰਿਕੁਐਸਟ ਪੂਰੀ ਹੋਣ 'ਤੇ ਬਟਨ ਦੁਬਾਰਾ ਚਾਲੂ btn.disabled = false; btn.innerText = "Classify"; } });
